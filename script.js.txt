// RENDER CHECKLIST
const checklistDiv = document.getElementById('checklist');
checklistItems.forEach(item=>{
  const key = item.toLowerCase().replace(/\s/g,'');
  checklistDiv.innerHTML += `<label><input type="checkbox" data-key="${key}"> ${item}</label>`;
});

// TAB SYSTEM
function showTab(id){
  document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));
  document.querySelectorAll('nav button').forEach(b=>b.classList.remove('active'));
  document.getElementById(id).classList.add('active');
  event.target.classList.add('active');
}

// SAVE CHECKLIST KE HP
document.querySelectorAll('input[type=checkbox]').forEach(cb=>{
  cb.checked = localStorage.getItem(cb.dataset.key) === 'true';
  cb.onchange = ()=>{localStorage.setItem(cb.dataset.key, cb.checked); updateProgress();}
});

function updateProgress(){
  const total = checklistItems.length;
  const done = document.querySelectorAll('#checklist input:checked').length;
  document.getElementById('progress').style.width = (done/total*100)+'%';
}

// RENDER JADWAL
function renderTable(id, data){
  const table = document.getElementById(id);
  table.innerHTML = '<tr><th>Jam</th><th>Aktivitas</th></tr>' +
    data.map(row=>`<tr><td>${row[0]}</td><td>${row[1]}</td></tr>`).join('');
}
renderTable('jadwal-fase1', jadwalFase1);
renderTable('tema-konten', temaKonten);

// RESET HARI
function resetHari(){
  if(confirm('Reset checklist?')){
    checklistItems.forEach(item=>{
      const key = item.toLowerCase().replace(/\s/g,'');
      localStorage.removeItem(key);
    });
    document.querySelectorAll('input[type=checkbox]').forEach(cb=>cb.checked=false);
    updateProgress();
  }
}

// SLEEP TRACKER
function addSleep(){
  const data = JSON.parse(localStorage.getItem('sleep')||'[]');
  data.push({tidur:'21:30',bangun:'05:00'});
  localStorage.setItem('sleep',JSON.stringify(data));
  loadSleep();
}
function loadSleep(){
  const data = JSON.parse(localStorage.getItem('sleep')||'[]');
  document.getElementById('sleep-list').innerHTML = data.map((d,i)=>`
    <div class="sleep-row">
      Hari ${i+1}:
      <input type="time" value="${d.tidur}" onchange="updateSleep(${i},'tidur',this.value)">
      <input type="time" value="${d.bangun}" onchange="updateSleep(${i},'bangun',this.value)">
    </div>
  `).join('');
}
function updateSleep(i,key,val){
  const data = JSON.parse(localStorage.getItem('sleep'));
  data[i][key]=val;
  localStorage.setItem('sleep',JSON.stringify(data));
}
if(!localStorage.getItem('sleep')) addSleep();
loadSleep();
updateProgress();
